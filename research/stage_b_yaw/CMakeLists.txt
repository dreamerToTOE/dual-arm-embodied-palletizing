cmake_minimum_required(VERSION 3.8)
project(stage_b_yaw)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ompl REQUIRED)
find_package(fcl REQUIRED)
find_library(LMA_PLUGIN moveit_lma_kinematics_plugin PATHS /opt/ros/humble/lib REQUIRED)
foreach(target yaw_planner alignment_diagnostic opposite_arm verify_replay)
  if(target STREQUAL alignment_diagnostic)
    add_executable(${target} ${DIAGNOSTIC_SOURCE})
  elseif(target STREQUAL opposite_arm)
    add_executable(${target} ${MIRROR_SOURCE})
  elseif(target STREQUAL verify_replay)
    add_executable(${target} verify_replay.cpp)
  else()
    add_executable(${target} yaw_planner.cpp)
  endif()
  target_include_directories(${target} PRIVATE ${OMPL_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../../offline/task28_stage_b_contact ${CMAKE_CURRENT_SOURCE_DIR}/../../ros_ws/src/fr3_dual_palletize/src)
  target_link_libraries(${target} ${LMA_PLUGIN} ${OMPL_LIBRARIES} yaml-cpp fcl)
  ament_target_dependencies(${target} rclcpp moveit_core moveit_msgs urdf srdfdom)
endforeach()

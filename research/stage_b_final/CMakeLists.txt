cmake_minimum_required(VERSION 3.8)
project(stage_b_final)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ompl REQUIRED)
find_package(fcl REQUIRED)
find_library(LMA_PLUGIN moveit_lma_kinematics_plugin PATHS /opt/ros/humble/lib REQUIRED)
set(GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/generate_sources.py ${GENERATED_DIR} RESULT_VARIABLE STATUS)
if(NOT STATUS EQUAL 0)
  message(FATAL_ERROR "source generation failed")
endif()
foreach(target precontact_left precontact_right yaw_left yaw_right)
  add_executable(${target} ${GENERATED_DIR}/${target}.cpp)
  target_include_directories(${target} PRIVATE ${OMPL_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../../offline/task28_stage_b_contact ${CMAKE_CURRENT_SOURCE_DIR}/../../ros_ws/src/fr3_dual_palletize/src)
  target_link_libraries(${target} ${LMA_PLUGIN} ${OMPL_LIBRARIES} yaml-cpp fcl)
  ament_target_dependencies(${target} rclcpp moveit_core moveit_msgs urdf srdfdom)
endforeach()
